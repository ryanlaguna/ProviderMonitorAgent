cmake_minimum_required(VERSION 3.20)
project(ProviderMonitorAgent VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX)
  message(FATAL_ERROR "This release requires Linux or macOS (POSIX subprocess support).")
endif()
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED cpp-httplib>=0.11)
find_path(HTTPLIB_INCLUDE_DIR httplib.h REQUIRED)
find_library(HTTPLIB_LIBRARY NAMES cpp-httplib REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(PMA_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(PMA_SQLITE_TARGET SQLite::SQLite3)
endif()
add_library(pma_core src/config.cpp src/process.cpp src/store.cpp src/agents.cpp)
target_include_directories(pma_core PUBLIC include)
target_link_libraries(pma_core PUBLIC CURL::libcurl ${PMA_SQLITE_TARGET} nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(pma_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(provider-monitor src/main.cpp)
target_include_directories(provider-monitor PRIVATE ${HTTPLIB_INCLUDE_DIR})
target_compile_options(provider-monitor PRIVATE ${HTTPLIB_CFLAGS_OTHER} -Wall -Wextra -Wpedantic)
target_link_libraries(provider-monitor PRIVATE pma_core ${HTTPLIB_LIBRARY})
configure_file(web/index.html ${CMAKE_CURRENT_BINARY_DIR}/web/index.html COPYONLY)
target_compile_definitions(provider-monitor PRIVATE PMA_WEB_DIR="${CMAKE_CURRENT_BINARY_DIR}/web")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME monitoring_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:provider-monitor>)
  set_tests_properties(monitoring_integration PROPERTIES TIMEOUT 120)
endif()
